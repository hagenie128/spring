import axios from 'axios';

const API_KEY = process.env.REACT_APP_TOUR_API_KEY;

export const fetchTourList = (params) =>
  axios.get('문제지에_적힌_URL', {
    params: {
      serviceKey: API_KEY,
      MobileOS: 'ETC',
      MobileApp: 'TestApp',
      _type: 'json',
      ...params,
    },
  });
